Minimum Window Substring — smallest substring in A containing all chars of B
🔥 HighSliding Window
▾

Sliding window with two frequency maps. Expand right, shrink left when valid. Track best window.

🔗 LC #76 
string minWindow(string s, string t) {
    unordered_map<char,int> need, window;
    for (char c : t) need[c]++;
    int have = 0, required = need.size();
    int l = 0, minLen = INT_MAX, start = 0;
    for (int r = 0; r < s.size(); r++) {
        window[s[r]]++;
        if (need.count(s[r]) && window[s[r]] == need[s[r]]) have++;
        while (have == required) {
            if (r - l + 1 < minLen) { minLen = r-l+1; start = l; }
            window[s[l]]--;
            if (need.count(s[l]) && window[s[l]] < need[s[l]]) have--;
            l++;
        }
    }
    return minLen == INT_MAX ? "" : s.substr(start, minLen);
}
// Time: O(|S|+|T|)  Space: O(|T|)

bool hasZeroSumSubarray(vector<int>& arr) {
    unordered_set<int> seen;
    seen.insert(0);
    int prefix = 0;
    for (int x : arr) {
        prefix += x;
        if (seen.count(prefix)) return true;

struct ListNode { int val; ListNode* next; };

ListNode* mergeKLists(vector<ListNode*>& lists) {
    auto cmp = [](ListNode* a, ListNode* b){ return a->val > b->val; };
    priority_queue<ListNode*, vector<ListNode*>, decltype(cmp)> pq(cmp);
    for (auto node : lists)
        if (node) pq.push(node);
    ListNode dummy(0); ListNode* cur = &dummy;
    while (!pq.empty()) {
        cur->next = pq.top(); pq.pop();
        cur = cur->next;
        if (cur->next) pq.push(cur->next);
    }
    return dummy.next;
}
// Time: O(N log K)  Space: O(K)

Jump Game II — minimum number of jumps (+ return jump indexes)
🔥 HighGreedy
▾

Greedy: track current range end and furthest reachable. When pointer hits current end, increment jump count.

🔗 LC #45 
int jump(vector<int>& nums) {
    int jumps=0, curEnd=0, farthest=0;
    for (int i=0; i < (int)nums.size()-1; i++) {
        farthest = max(farthest, i + nums[i]);
        if (i == curEnd) { jumps++; curEnd = farthest; }
    }
    return jumps;
}
// Time: O(N)  Space: O(1)

// Variant: also return the jump indexes
vector<int> jumpWithPath(vector<int>& nums) {
    int n = nums.size();
    vector<int> prev(n, -1);
    int curEnd=0, farthest=0, jumpFrom=0;
    for (int i=0; i<n-1; i++) {
        if (i + nums[i] > farthest) { farthest = i+nums[i]; jumpFrom = i; }
        if (i == curEnd) { prev[farthest] = jumpFrom; curEnd = farthest; }
    }
    vector<int> path;
    for (int i=n-1; i>0; i=prev[i]) path.push_back(i);
    path.push_back(0); reverse(path.begin(), path.end());
    return path;
}

string multiply(string num1, string num2) {
    int m=num1.size(), n=num2.size();
    vector<int> pos(m+n, 0);
    for (int i=m-1; i>=0; i--)
        for (int j=n-1; j>=0; j--) {
            int mul = (num1[i]-'0')*(num2[j]-'0');
            int p1=i+j, p2=i+j+1, sum=mul+pos[p2];
            pos[p2]=sum%10; pos[p1]+=sum/10;
        }
    string res;
    for (int d : pos) if (!(res.empty() && d==0)) res+=('0'+d);
    return res.empty() ? "0" : res;
}
// Time: O(M*N)  Space: O(M+N)

int maxDistance(vector<int>& arr) {
    int n=arr.size();
    vector<int> lmin(n), rmax(n);
    lmin[0]=arr[0];
    for (int i=1;i<n;i++) lmin[i]=min(lmin[i-1],arr[i]);
    rmax[n-1]=arr[n-1];
    for (int j=n-2;j>=0;j--) rmax[j]=max(rmax[j+1],arr[j]);
    int i=0, j=0, ans=0;
    while (i<n && j<n) {
        if (lmin[i]<=rmax[j]) { ans=max(ans,j-i); j++; }
        else i++;
    }
    return ans;
}
// Time: O(N)  Space: O(N)

vector<int> sortByFreq(vector<int>& arr) {
    map<int,int> freq;
    for (int x : arr) freq[x]++;
    sort(arr.begin(), arr.end(), [&](int a, int b){
        if (freq[a] != freq[b]) return freq[a] < freq[b];
        return a > b; // tie-break by value desc
    });
    return arr;
}

vector<int> searchRange(vector<int>& nums, int target) {
    auto bound = [&](bool lower) {
        int lo=0, hi=nums.size()-1, res=-1;
        while(lo<=hi){
            int mid=lo+(hi-lo)/2;
            if(nums[mid]==target){ res=mid; lower?hi=mid-1:lo=mid+1; }
            else if(nums[mid]<target) lo=mid+1;
            else hi=mid-1;
        }
        return res;
    };
    return {bound(true), bound(false)};
}
// Time: O(log N)  Space: O(1)

ListNode* reverseKGroup(ListNode* head, int k) {
    ListNode* cur=head; int cnt=0;
    while(cur && cnt<k){ cur=cur->next; cnt++; }
    if(cnt<k) return head;
    ListNode* prev=nullptr; cur=head;
    for(int i=0;i<k;i++){
        ListNode* nxt=cur->next; cur->next=prev; prev=cur; cur=nxt;

    return prev;
}

Node* connect(Node* root) {
    if(!root) return nullptr;
    Node* leftmost=root;
    while(leftmost->left){
        Node* head=leftmost;
        while(head){
            head->left->next=head->right;
            if(head->next) head->right->next=head->next->left;
            head=head->next;
        }
        leftmost=leftmost->left;

    return root;
}

LCA / Common Manager of Two Employees
🔥 HighTree
▾

Disguised as "common manager" — it's LCA on a general tree. Key insight: if the Emp class has a parent pointer, walk both nodes to root, collect ancestors, find first match.

🔗 LC #236 
struct TreeNode{int val;TreeNode*left,*right;};

// Binary Tree LCA (no parent pointer)
TreeNode* lca(TreeNode* root, TreeNode* p, TreeNode* q) {
    if(!root||root==p||root==q) return root;
    TreeNode* L=lca(root->left,p,q);
    TreeNode* R=lca(root->right,p,q);
    if(L && R) return root;
    return L ? L : R;
}
// Time: O(N)  Space: O(H)

// With parent pointer (closer to real "Emp" class question)
struct Emp { int id; Emp* manager; };
Emp* commonManager(Emp* e1, Emp* e2) {
    unordered_set<Emp*> ancestors;
    while(e1){ ancestors.insert(e1); e1=e1->manager; }
    while(e2){ if(ancestors.count(e2)) return e2; e2=e2->manager; }
    return nullptr;
}
// Time: O(D)  Space: O(D) where D = depth

vector<vector<int>> verticalTraversal(TreeNode* root) {
    map<int,map<int,multiset<int>>> nodes;
    queue<tuple<TreeNode*,int,int>> q;
    q.push({root,0,0});
    while(!q.empty()){
        auto[node,col,row]=q.front(); q.pop();
        nodes[col][row].insert(node->val);
        if(node->left) q.push({node->left,col-1,row+1});
        if(node->right) q.push({node->right,col+1,row+1});
    }
    vector<vector<int>> res;
    for(auto&[col,rows]:nodes){
        res.push_back({});
        for(auto&[row,vals]:rows)
            for(int v:vals) res.back().push_back(v);
    }
    return res;
}

bool dfs(vector<vector<int>>& g, int r, int c, int dr, int dc) {
    if(r==dr && c==dc) return true;
    int n=g.size(), m=g[0].size();
    int dirs[][2]={{1,0},{-1,0},{0,1},{0,-1}};
    g[r][c]=0; // mark visited
    for(auto& d:dirs){
        int nr=r+d[0],nc=c+d[1];
        if(nr>=0&&nr<n&&nc>=0&&nc<m&&g[nr][nc]==1)
            if(dfs(g,nr,nc,dr,dc)) return true;
    }
    g[r][c]=1; // backtrack
    return false;
}
// Time: O(4^(N*M))  Space: O(N*M)

int timer=0;
void dfs(vector<vector<int>>& g, int u, int par,
         vector<int>& disc, vector<int>& low,
         vector<vector<int>>& res) {
    disc[u]=low[u]=timer++;
    for(int v:g[u]){
        if(disc[v]==-1){
            dfs(g,v,u,disc,low,res);
            low[u]=min(low[u],low[v]);
            if(low[v]>disc[u]) res.push_back({u,v});
        } else if(v!=par) low[u]=min(low[u],disc[v]);
    }
}
vector<vector<int>> criticalConnections(int n, vector<vector<int>>& edges) {
    vector<vector<int>> g(n), res;
    vector<int> disc(n,-1), low(n);
    for(auto& e:edges){ g[e[0]].push_back(e[1]); g[e[1]].push_back(e[0]); }
    for(int i=0;i<n;i++) if(disc[i]==-1) dfs(g,i,-1,disc,low,res);
    return res;
}

vector<int> dijkstra(vector<vector<pair<int,int>>>& g, int src){
    int n=g.size();
    vector<int> dist(n,INT_MAX);
    priority_queue<pair<int,int>,vector<pair<int,int>>,greater<>> pq;
    dist[src]=0; pq.push({0,src});
    while(!pq.empty()){
        auto[d,u]=pq.top(); pq.pop();
        if(d>dist[u]) continue;
        for(auto[v,w]:g[u])
            if(dist[u]+w<dist[v]){ dist[v]=dist[u]+w; pq.push({dist[v],v}); }
    }
    return dist;
}
// Time: O((V+E) log V)  Space: O(V)

    if(l==s.size()-1){ res.insert(s); return; }

long long diceThrow(int D, int F, int S){
    // dp[i][j] = ways to get sum j using i dice

string longestRepeating(string s){
    int n=s.size();
    vector<vector<int>> dp(n+1,vector<int>(n+1,0));
    int len=0, idx=0;
    for(int i=1;i<=n;i++)
        for(int j=i+1;j<=n;j++)
            if(s[i-1]==s[j-1] && dp[i-1][j-1]<j-i){
                dp[i][j]=dp[i-1][j-1]+1;
                if(dp[i][j]>len){ len=dp[i][j]; idx=i-1; }
            }

☕ Core Java Internals ▾
What is a Record in Java? How is it different from a POJO?
A record (Java 16+) is a concise immutable data carrier. The compiler auto-generates constructor, getters, equals(), hashCode(), and toString(). Fields are implicitly private final. A POJO requires you to write all of these manually and fields can be mutable. Records are ideal for DTOs and value objects where immutability is desired.

Example: record Point(int x, int y) {} — that's the entire class.
Why is the default capacity of HashMap 16? Why must it be a power of 2?
The bucket index is computed as hash & (capacity - 1) instead of hash % capacity. This bitwise AND is only equivalent to modulo when capacity is a power of 2, and it is significantly faster. Starting at 16 provides a reasonable balance — small enough not to waste memory for small maps, large enough to avoid immediate rehashing. Load factor 0.75 means rehashing at 12 entries.
What are volatile, synchronized, and AtomicInteger? When to use each?
  • volatile: Guarantees visibility — writes are immediately flushed to main memory, reads always fetch from main memory. Does NOT guarantee atomicity for compound operations like i++. Use for simple flags (volatile boolean running).
  • synchronized: Guarantees both visibility and atomicity but is blocking. Use for complex critical sections where multiple operations must be atomic together.
  • AtomicInteger: Non-blocking (CAS — Compare and Swap). Guarantees atomicity for single-variable operations without locking. Use for counters, reference updates. Preferred over synchronized for single variable.
Explain equals() and hashCode() contract. What breaks if you override only one?
Contract: if a.equals(b) is true, then a.hashCode() == b.hashCode() must be true. If you override equals() but not hashCode(): two logically equal objects get different hash codes → they land in different HashMap buckets → map.get() returns null even after put(). If you override only hashCode(): objects with the same hash could be considered unequal → duplicate keys in collections.
What are Java 8 Streams? Why use them over for-loops?
Streams provide a declarative, functional-style pipeline for processing collections: filter → map → reduce. Benefits: (1) more readable code, (2) lazy evaluation — intermediate ops only execute when terminal op is called, (3) easy parallelism via parallelStream() with no manual thread management, (4) composable operations. Downside: harder to debug, slight overhead for small collections. Use streams for data transformation pipelines; use loops for performance-critical hot paths.
How does ExecutorService, CompletableFuture, and ForkJoinPool differ?
  • ExecutorService: Thread pool manager. Submit Callable/Runnable tasks, get Future back. Blocking future.get() to retrieve result.
  • CompletableFuture: Non-blocking composition of async tasks. Chain with .thenApply(), .thenCompose(), .exceptionally(). Uses ForkJoinPool.commonPool() by default or a custom executor.
  • ForkJoinPool: Work-stealing pool designed for recursive divide-and-conquer tasks. Best for CPU-bound recursive algorithms. parallelStream() uses it internally.
🌱 Spring Boot / Spring Framework ▾
What is an Interceptor in Spring? Why use it over a Filter?
A Spring HandlerInterceptor intercepts requests at the Spring MVC level — after the request reaches DispatcherServlet but before it hits the controller (preHandle), after the controller (postHandle), and after view rendering (afterCompletion). A Servlet Filter operates at the Servlet container level — before Spring even processes the request. Use Interceptors for Spring-specific logic (accessing ModelAndView, accessing Spring beans easily). Use Filters for low-level request/response manipulation (CORS, compression, security headers) that must apply to all endpoints including non-Spring ones.

Example use case: logging request duration, authentication token validation, audit logging.
Explain @Transactional propagation = REQUIRED vs REQUIRES_NEW
  • REQUIRED (default): Join existing transaction if one exists; create new one if not. Both methods succeed or fail together.
  • REQUIRES_NEW: Always create a new transaction, suspend any existing one. The inner transaction commits/rolls back independently. Use for audit logging that must persist even if the outer transaction fails.
Common pitfall: @Transactional only works on public methods called from outside the bean. Self-invocation (calling a @Transactional method from the same class) bypasses the proxy — the transaction is ignored.
How does Spring Boot auto-configuration work?
@EnableAutoConfiguration triggers class-path scanning via spring.factories (or AutoConfiguration.imports in Spring Boot 3). Each auto-configuration class is annotated with @ConditionalOnClass, @ConditionalOnMissingBean, etc. For example, if spring-data-redis is on classpath, RedisAutoConfiguration auto-creates a RedisTemplate bean unless you define your own. You can exclude specific auto-configs via @SpringBootApplication(exclude = {DataSourceAutoConfiguration.class}).
What is AOP? Give a real production use case.
Aspect-Oriented Programming allows cross-cutting concerns to be modularised out of business logic. Spring implements it via proxies (JDK dynamic proxy for interfaces, CGLIB for classes). Common production use cases: (1) Logging: log method entry/exit/args without polluting business code. (2) Performance monitoring: measure execution time and push to Micrometer/Prometheus. (3) Security: @PreAuthorize. (4) Transaction management: @Transactional is itself AOP. (5) Rate limiting: intercept annotated methods and check rate limit bucket.

Key terms: @Aspect, @Before, @After, @Around, @Pointcut.

How does Spring Security JWT authentication work end-to-end?
1. User POSTs credentials → AuthController → AuthenticationManager.authenticate() → UserDetailsService.loadByUsername() → verify password → generate JWT (signed with secret, contains userId + roles + expiry).
2. Client sends Authorization: Bearer <token> on every request.
3. JwtAuthFilter (extends OncePerRequestFilter) extracts token, validates signature + expiry, sets SecurityContextHolder.getContext().setAuthentication().

How do you prevent double payment when the same request is made twice? (Explicitly asked at MMT)

How do you handle concurrent seat booking? (IRCTC-style — asked multiple times)

Then UPDATE seat SET status='BOOKED', version=version+1 WHERE id=? AND version=?. If 0 rows updated → conflict → retry or show "seat taken". What is the Circuit Breaker pattern? How does Resilience4j implement it?
Circuit Breaker prevents cascading failures by stopping calls to a failing downstream service. Three states:
  • CLOSED: Normal. Calls pass through. Failure rate tracked in a sliding window (count or time-based).
  • OPEN: Failure threshold exceeded. All calls immediately fail (or return fallback) without hitting the service. Timer runs.
  • HALF_OPEN: Timer elapsed. Limited calls permitted. If they succeed → CLOSED. If they fail → back to OPEN.
In Spring Boot: @CircuitBreaker(name="payment", fallbackMethod="fallback"). Config: failureRateThreshold=50, waitDurationInOpenState=10s, permittedNumberOfCallsInHalfOpenState=5.
Kafka — Topics, Partitions, Consumer Groups, Offset management, Exactly-once
  • Topic: Named stream of records. Divided into Partitions for parallelism. Messages within a partition are ordered.
  • Consumer Group: Multiple consumers sharing load. Each partition is consumed by exactly one consumer in a group. Scaling: add more consumers up to the number of partitions.
  • Offset: Position within a partition. Auto-commit (at-least-once — may reprocess on crash) vs manual commit (control exactly when offset advances). Commit after successful processing for at-least-once.
  • Exactly-once: Enable idempotent producer (enable.idempotence=true) + Kafka transactions (transactional.id) + isolation.level=read_committed on consumer. Or handle at application level with idempotency keys.
How do you reduce latency between two microservices? (Tech Lead core question Jul 2024)
Multi-pronged approach depending on root cause:
  • Async decoupling: If response isn't needed synchronously, use Kafka/SQS. Service A publishes event, Service B consumes. Eliminates synchronous wait.
  • Caching: Redis read-through cache in Service A for data fetched from Service B. Reduces cross-service calls for static/semi-static data.
  • gRPC instead of REST: Binary protocol (protobuf), HTTP/2 multiplexing, persistent connections. 5–10x faster than REST for internal service calls.
  • Connection pooling: Use Apache HttpClient with connection pool instead of creating new connections per request.
  • Bulk/Batch APIs: Replace N calls with 1 batch call (e.g. fetch 100 prices in one request).
  • Co-location: Deploy both services in the same availability zone/region to minimise network hop latency.
What is the Saga Pattern? Choreography vs Orchestration?
Saga manages distributed transactions across microservices where 2PC is not feasible. Each step has a compensating transaction for rollback.

Choreography: Each service publishes events and reacts to others' events (event-driven). No central coordinator. Pros: loose coupling. Cons: hard to track flow, distributed logic.

Orchestration: A central Saga Orchestrator (e.g. Temporal, or a dedicated Spring service) calls each service in sequence and handles compensation on failure. Pros: clear flow, easier debugging. Cons: orchestrator is a central dependency.

MMT context: Booking saga — reserve seat → charge payment → send confirmation email. If payment fails → release seat (compensation).
💾 Databases — MySQL, Redis, NoSQL ▾
SQL vs NoSQL — when would you choose MongoDB over MySQL?
Choose MySQL (relational) when: data is structured and relational (users, orders, payments), ACID transactions are required (financial data), complex JOINs needed, schema is stable.

Choose MongoDB (NoSQL) when: data is hierarchical/nested (hotel metadata with room types, amenities, photos as nested arrays), schema varies across documents (different hotel types have different fields), horizontal sharding needed for massive write scale, rapid schema iteration during product development.

At MMT: booking and payment tables → MySQL. Hotel/flight content catalogue (deeply nested, varies by type) → MongoDB.
How do indexes work? How does a B-tree index work? When does an index hurt performance?
A B-tree index is a balanced tree structure stored separately from the table. Each leaf node contains the indexed value + a pointer to the row (or the clustered key). Lookups: O(log N). Range queries: O(log N + K) — tree traversal then leaf-level scan.

Indexes hurt when: (1) Write-heavy tables: every INSERT/UPDATE/DELETE must also update all indexes. (2) Low-cardinality columns: index on a boolean column with 2 values is useless — MySQL will do a full table scan anyway. (3) Too many indexes: optimizer must choose among many; storage overhead. (4) Functions on indexed columns: WHERE YEAR(created_at) = 2024 can't use an index on created_at — rewrite as range query.
What are Redis data structures and when to use each?
  • String: Simple key-value, counters (INCR), rate limiting, session tokens, feature flags.
  • Hash: User/session objects — store fields without serializing entire JSON. HSET user:1 name "Ashutosh" age 27.
  • List: Message queues (LPUSH/BRPOP), activity feeds, recent items.
  • Set: Unique tags, friend lists, deduplication, set operations (SUNION, SINTER).
  • Sorted Set (ZSet): Leaderboards (score = timestamp or points), rate limiting (sliding window log with timestamps as scores), geospatial indexes.
  • HyperLogLog: Approximate distinct count (DAU) with ~1% error, fixed 12KB memory regardless of cardinality.
Cache invalidation strategies — Write-through vs Write-behind vs Cache-aside?
  • Cache-aside (Lazy loading): App checks cache first. Miss → fetch from DB → populate cache. Pros: only caches what's needed. Cons: initial miss penalty, potential stale data. Most common pattern.
  • Write-through: Every write goes to cache AND DB synchronously. Pros: cache always fresh. Cons: writes are slower (two writes), caches data that may never be read.
  • Write-behind (Write-back): Write to cache immediately, persist to DB asynchronously. Pros: fastest writes. Cons: risk of data loss if cache crashes before DB write. Use only where eventual persistence is acceptable.
Cache-aside is MMT's likely pattern for hotel/flight search results with configurable TTLs.

🔧 OS & Computer Networks ▾
Process vs Thread — difference? Why is context switching expensive for processes?
A process is an independent execution unit with its own memory space, file handles, and OS resources. A thread is a lightweight execution unit within a process, sharing the same heap/code/data segments.

Context switching is more expensive for processes because: (1) full memory address space must be saved/restored including page tables, (2) TLB (Translation Lookaside Buffer) must be flushed — next process needs its own virtual-to-physical mapping, (3) more kernel data structures. Thread switching only saves/restores CPU registers and stack pointer — no TLB flush needed within the same process.
What is a deadlock? Coffman's conditions. How to prevent in Java?
Deadlock: two or more threads wait indefinitely for each other to release locks. Coffman's 4 conditions (all must hold): (1) Mutual Exclusion, (2) Hold & Wait, (3) No Preemption, (4) Circular Wait.

Prevention in Java: (1) Lock ordering: always acquire locks in the same order across all threads. (2) tryLock with timeout: lock.tryLock(1, TimeUnit.SECONDS) — give up and retry rather than wait forever. (3) Avoid nested locks: redesign to avoid needing two locks simultaneously. (4) Use higher-level constructs: java.util.concurrent classes (ConcurrentHashMap, BlockingQueue) which handle their own synchronization safely.
HTTP vs HTTPS — what happens during TLS handshake?
1. Client sends ClientHello: TLS version, supported cipher suites, random bytes.
2. Server sends ServerHello: chosen cipher suite, random bytes, SSL Certificate (contains server's public key, signed by CA).
3. Client verifies certificate against trusted CA store.
4. Key exchange: (TLS 1.3) Client/Server use ECDHE to derive a shared secret without transmitting it — perfect forward secrecy.
5. Both derive the same symmetric session key from the shared secret + random bytes.
6. All subsequent data encrypted with symmetric key (e.g. AES-256-GCM).

HTTPS adds confidentiality (encryption), integrity (HMAC), and authentication (certificate) over HTTP.
What happens when you type "www.google.com" in a browser?
1. DNS Resolution: Browser cache → OS cache → Recursive resolver → Root NS → .com TLD NS → Google's authoritative NS → IP address returned.
2. TCP Handshake: SYN → SYN-ACK → ACK to the resolved IP on port 443.
3. TLS Handshake: As described above — establishes encrypted channel.
4. HTTP/2 Request: GET / HTTP/2 with headers (Host, Accept, Cookie).
5. Server Processing: Load balancer → web server → application server → (cache → DB if needed) → response built.
6. Response: HTML sent back → browser parses → fetches CSS/JS/images (parallel HTTP/2 streams) → renders page.

class RateLimiter {
    private final int maxRequests;
    private final long windowMs;
    private final Map<String, Deque<Long>> userTimestamps = new ConcurrentHashMap<>();

    RateLimiter(int max, long windowMs) { this.maxRequests=max; this.windowMs=windowMs; }

    public synchronized boolean isAllowed(String userId) {
        long now = System.currentTimeMillis();
        userTimestamps.putIfAbsent(userId, new ArrayDeque<>());
        Deque<Long> timestamps = userTimestamps.get(userId);
        // evict old timestamps outside window
        while (!timestamps.isEmpty() && now - timestamps.peekFirst() > windowMs)
            timestamps.pollFirst();
        if (timestamps.size() < maxRequests) {
            timestamps.addLast(now);
            return true;   // allowed
        }
        return false;   // rate limited → return 429
    }
}

    @Autowired RedisTemplate<String,String> redis;

    public String shorten(String longUrl, Long userId) {
        String code = toBase62(repo.save(new UrlMapping(longUrl, userId)).getId());
        redis.opsForValue().set("url:"+code, longUrl, 7, TimeUnit.DAYS);
        return "https://mmt.to/" + code;
    }

    public String resolve(String code, Long requestUserId) {
        String cached = redis.opsForValue().get("url:"+code);
        if (cached != null) return cached;
        UrlMapping mapping = repo.findByShortCode(code)
            .orElseThrow(() -> new NotFoundException());
        if (!mapping.getUserId().equals(requestUserId)) throw new ForbiddenException();
        return mapping.getLongUrl();
    }

    private String toBase62(long n) {
        StringBuilder sb = new StringBuilder();
        while (n > 0) { sb.append(CHARS.charAt((int)(n%62))); n/=62; }
        return sb.reverse().toString();
    }
}

    }

        synchronized(acc) {
            if (acc.getBalance() < amount) throw new InsufficientFundsException();
            acc.setBalance(acc.getBalance() - amount);
        }

        // Lock in consistent order to prevent deadlock
        Account first  = fromId < toId ? getOrThrow(fromId) : getOrThrow(toId);
        Account second = fromId < toId ? getOrThrow(toId)   : getOrThrow(fromId);
        synchronized(first) {
            synchronized(second) {
                withdraw(fromId, amount);
                deposit(toId, amount);
            }
        }
    }

#2
Idempotent Booking + Double Payment Prevention
🔥 HighHLD
▾
Full flow:
1. Client generates UUID idempotencyKey per booking attempt.
2. API: POST /bookings with header Idempotency-Key: uuid.
3. Server atomically SET booking:idem:{key} PROCESSING NX EX 300 in Redis.
4. If key already exists → check if value is DONE:{bookingId} → return cached response (200).
5. Process booking → charge payment → store result as DONE:{bookingId} in Redis with 24h TTL.
6. Also store idempotency_key as UNIQUE column in Booking table — second layer of safety.

For the payment charge itself: Payment gateway also supports idempotency keys. Pass the same key so even if you call the payment gateway twice (network retry), only one charge occurs.

Outbox Pattern (for reliable event publishing): Write booking + outbox event in same DB transaction. Separate poller reads outbox and publishes to Kafka. Guarantees event is published exactly once even if service crashes.

#4
Distributed Cache Design
⚡ MedHLD
▾
Key components:
1. Consistent Hashing: Map keys to nodes on a ring. Adding/removing a node only remaps ~K/N keys (vs. K/2 with simple modulo). Use virtual nodes for uniform distribution.
2. Replication: Each key stored on N nodes (e.g., 3). Reads from any replica. Writes to primary → async replication.
3. Eviction: LRU per shard with max memory limit. TTL expiry (lazy + active background sweep).
4. Cache Coherence: On write, invalidate all replicas (delete-on-write) or use versioned values (read-your-writes consistency).
5. Client-side routing: Clients use the same consistent hash ring to determine which node to contact — no central router bottleneck.

Redis Cluster in production implements this with 16384 hash slots distributed across master nodes, each with replicas.
